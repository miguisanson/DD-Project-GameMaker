    _s.lay = _l;

    display_set_gui_size(-1, -1);         // reset any earlier size

    var _fe = 1, _ko_pt = 11, _ko_k = 1;   // english scale (GUI units), korean pt and korean draw scale
    var _n = max(1, round(0.8 * _sc));

function Spike_Step() {
    if (!variable_global_exists("spk")) return;
    var _s = global.spk;
    if (_s.done) return;
    _s.f++;
    with (obj_sfx_manager) depth = 1000000;
    if (_s.f > 6000) { Spk_Log("TIMEOUT"); game_end(1); _s.done = true; return; }
    if (_s.cap != "" && !_s.cap_done && _s.cap_via == "step") {
        var _w = window_get_width(), _h = window_get_height();
        screen_save(_s.cap + ".png");
        Spk_Log("[SHOT] file=" + _s.cap + ".png w=" + string(_w) + " h=" + string(_h) + " via=step");
        _s.cap_done = true;
        return;
    }
    if (_s.lagn > 0) {
        var _ap = application_get_position();
        Spk_Log("lag t+" + string(13 - _s.lagn) + " win=" + string(window_get_width()) + "x" + string(window_get_height()) + " gui=" + string(display_get_gui_width()) + "x" + string(display_get_gui_height()) + " appPos=" + string(_ap[0]) + "," + string(_ap[1]) + "," + string(_ap[2]) + "," + string(_ap[3]));
        _s.lagn--;
    }
    if (!_s.cap_done) return;           // waiting for a capture
    if (_s.wait_file != "") {
        _s.wait_n++;
        if (file_exists(_s.wait_file)) { file_delete(_s.wait_file); _s.wait_file = ""; }
        else if (_s.wait_n > 1200) { Spk_Log("grab ack TIMEOUT " + _s.wait_file); _s.wait_file = ""; }
        return;
    }
    if (_s.win_wait > 0) {              // waiting for window_set_size to land
        if (window_get_width() == _s.win_target[0] && window_get_height() == _s.win_target[1]) {
            Spk_Log("window reached " + string(_s.win_target[0]) + "x" + string(_s.win_target[1]) + " after " + string(_s.f - _s.win_t0) + " frames");
            _s.win_wait = -3;           // 3 more frames
        } else if (_s.f - _s.win_t0 > 120) {
            Spk_Log("window NEVER reached " + string(_s.win_target[0]) + "x" + string(_s.win_target[1]) + ", is " + string(window_get_width()) + "x" + string(window_get_height()));
            _s.win_wait = 0;
        }
        return;
    }
    if (_s.win_wait < 0) { _s.win_wait++; return; }
    if (_s.t > 0) { _s.t--; return; }
    if (_s.step >= array_length(_s.ops)) return;
    var _o = _s.ops[_s.step];
    _s.step++;
    switch (_o.op) {
        case "wait": _s.t = _o.n; break;
        case "rep": Spk_State(_o.tag); break;
        case "win":
            Spk_Log("window_set_size " + string(_o.w) + "x" + string(_o.h) + " now=" + string(window_get_width()) + "x" + string(window_get_height()));
            window_set_size(_o.w, _o.h);
            Spk_Log("  immediately after: window_get=" + string(window_get_width()) + "x" + string(window_get_height()));
            _s.win_target = [_o.w, _o.h]; _s.win_t0 = _s.f; _s.win_wait = 1;
            break;
        case "winraw": window_set_size(_o.w, _o.h); _s.win_target = [_o.w, _o.h]; _s.win_t0 = _s.f; _s.win_wait = 1; break;
        case "layout": Spk_Apply(_o.mode); break;
        case "lag":
            Spk_Log("lag: set_size " + string(_o.w) + "x" + string(_o.h) + " from win=" + string(window_get_width()) + " gui=" + string(display_get_gui_width()) + "x" + string(display_get_gui_height()) + " appPos=" + string(application_get_position()[0]));
            window_set_size(_o.w, _o.h); _s.lagn = 12; break;
        case "hideui": with (obj_start_controller) visible = false; with (obj_ui_controller) visible = false; Spk_Log("hid game ui"); break;
        case "grab": Spk_Log("GRAB " + _o.name + " win=" + string(window_get_width()) + "x" + string(window_get_height())); _s.wait_file = "ack_" + _o.name; _s.wait_n = 0; break;
        case "cap": _s.cap = _o.name; _s.cap_via = _o.via; _s.cap_done = false; break;
        case "goto":
            if (_o.r == rm_floor1 || _o.r == rm_battle) { Scenario_NewKnight(); GameState_SetBattleReturn(rm_floor1, 0, 0, 0); GameState_SetBattleEnemy("spk", ENEMY_SLIME, 1); }
            room_goto(_o.r);
            break;
        case "ui": _s.ui = _o.rule; Spk_Log("ui rule " + _o.rule); break;
        case "end":
            Spk_Log("[TEST-SUMMARY] pass=1 fail=0 errors=0");
            _s.done = true;
            game_end(0);
            break;
    }
}
